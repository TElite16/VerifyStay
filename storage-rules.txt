rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Verification documents (utility bill, land doc, etc.)
    // Only the owner can upload into their own folder. Reading requires
    // login, since these are sensitive personal documents — not for
    // public display, only for the uploader and admin review.
    match /verification-docs/{userId}/{fileName} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if request.auth != null
                   && request.auth.uid == userId
                   && request.resource.size < 10 * 1024 * 1024
                   && (request.resource.contentType.matches('image/.*')
                       || request.resource.contentType == 'application/pdf');
    }

    // Property photos — publicly viewable (they're marketing images
    // for a public listing), but only the owner can upload into their
    // own folder.
    match /properties/{userId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
                   && request.auth.uid == userId
                   && request.resource.size < 10 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }

    // Default deny — anything not explicitly listed above is blocked.
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
